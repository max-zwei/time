Data Analysis¶

Context¶

For the module "Infographics" I"m trying to analyse data regarding the time usage of people aged 15 - 29 year old, reflecting the typical age range at CODE University of Applied Sciences.

Problem question¶

How do teenagers spent their time on a daily basis? And how did that change from 1991 to 2001 to 2012 to 2022? And what can be derived to the impact of smartphones and social media?

Data¶

Forschungsdatenzentren Der Statistischen Ämter Des Bundes Und Der Länder. (2018a). Zeitbudgeterhebung 1991/1992, PUF, Version 0 (Version 0) [Data set]. Research Data Centres of the Statistical Offices of the Federation and the Federal States. https://doi.org/10.21242/63911.1992.00.00.4.1.0

Forschungsdatenzentren Der Statistischen Ämter Des Bundes Und Der Länder. (2018b). Zeitbudgeterhebung 2001/2002, PUF, Version 0 (Version 0) [Data set]. Research Data Centres of the Statistical Offices of the Federation and the Federal States. https://doi.org/10.21242/63911.2002.00.00.4.1.0

Forschungsdatenzentren Der Statistischen Ämter Des Bundes Und Der Länder. (2018c). Zeitverwendungserhebung 2012/2013, PUF, Version 0 (Version 0) [Data set]. Research Data Centres of the Statistical Offices of the Federation and the Federal States. https://doi.org/10.21242/63911.2013.00.00.4.1.0

Forschungsdatenzentren Der Statistischen Ämter Des Bundes Und Der Länder. (2018d). Zeitverwendungserhebung 2022, PUF, Version 0 (Version 0) [Data set]. Research Data Centres of the Statistical Offices of the Federation and the Federal States. https://doi.org/10.21242/63911.2022.00.00.4.1.1

Deliverables¶

The aim of this data analysis is to find patterns and trends in how the spent time changed from 2001 to 2001, to 2012 and to finally 2022. To achieve that, the data analysis should provide as a first deliverable the following data table:

  • activity_id
  • activity_label
  • year
  • people
  • mean
  • median
  • std
  • mean+2std (as of 2001)

# Delete the summed activity code groups, like ha_2 or ha_20

def keep_detailed_activities(dataframe):
    columns_to_keep = ["weights", "pers_id"]

    for column in dataframe.columns:
        # Skip columns that aren't activities and save activites as code.
        if column.startswith("ZH"):
            code = column.replace("ZH", "")
        elif column.startswith("ha_"):
            code = column.replace("ha_", "")
        else:
            continue

        # Some activity codes have an extra "x" at the end.
        code = code.removesuffix("x")

        # Keep detailed activities, which have three digits.
        if len(code) == 3 and code.isdigit():
            columns_to_keep.append(column)

    return dataframe.select(columns_to_keep)

The mapping includes some best guesses as not everything could be mapped to a corresponding code for the following year. In extreme cases these remain with the activity code "UNRESOLVED" to keep the data comparison as clean as possible.

## Map codes
# 1 to 1 translations
# 1 to X translations
# 1 to "UNRESOLVED: ..." translations

def map_codes(df_map, df_data, mapping_year):
    # Create a lookup: old activity code → new activity code(s).
    mapping = {}

    for old_code, new_code in df_map.iter_rows():
        mapping[old_code.strip()] = new_code

    # Start the result with only the person IDs and weights.
    result = df_data.select("pers_id", "weights")

    for column in df_data.columns:
        if column == "pers_id" or column == "weights":
            continue

        # Keep codes that were already marked unresolved.
        if column.startswith("UNRESOLVED"):
            new_code = column
        else:
            # 2012 doesn't list all codes
            if mapping_year == 2012:
                new_code = mapping.get(column, column)
            else:
                new_code = mapping.get(column)

            if new_code is None or new_code.strip() in ["", "na", "UNRESOLVED"]:
                new_code = f"UNRESOLVED:{mapping_year}:{column}"

        # 1 to x translations with an old code mapped to multiple new codes divided by "/", e.g. "ha_141/ha_142"
        destinations = new_code.split("/")

        # Divide this activity's minutes equally between them for the new codes.
        minutes_per_destination = df_data[column] / len(destinations)

        for destination in destinations:
            destination = destination.strip()

            if destination in result.columns:
                # Another activity has already contributed to this code.
                combined_minutes = (
                    result[destination] + minutes_per_destination
                )
                result = result.with_columns(
                    combined_minutes.alias(destination)
                )
            else:
                result = result.with_columns(
                    minutes_per_destination.alias(destination)
                )

    return result

# Group activities for levels 1 and 2

def group_activities(dataframe, level):
    result = dataframe.select("pers_id", "weights")

    for column in dataframe.columns:
        if column in ["pers_id", "weights"]:
            continue

        if column.startswith("UNRESOLVED"):
            group_code = "UNRESOLVED"
        else:
            digits = column.replace("ha_", "")
            digits = digits.removesuffix("x")

            if len(digits) < level:
                group_code = "UNRESOLVED:GROUP:" + column
            else:
                group_code = "ha_" + digits[:level]

        # Get this activity's minutes for every diary.
        minutes = dataframe[column]

        if group_code in result.columns:
            # Add to a group we have already started.
            minutes = result[group_code] + minutes

        result = result.with_columns(
            minutes.alias(group_code)
        )

    return result

# Create a nested list with the activity code as row and the required columns: [[ha_1, year, [...]] 
def activity_descriptions(dataframe, weights_col, year):
    rows = []
    weights_values = dataframe.get_column(weights_col).to_numpy()

    # number of people for that year
    n_people = dataframe["pers_id"].unique().len()

    for column in dataframe.columns:
        if column in {weights_col, "pers_id"}:
            continue 

        values = dataframe.get_column(column).to_numpy()

        #sum
        sum = np.sum(values * weights_values)

        # mean  
        mean = np.average(values, weights=weights_values)

        # median
        order = np.argsort(values)
        sorted_values = values[order]
        sorted_weights = weights_values[order]

        cumulative_weights = np.cumsum(sorted_weights)
        halfway = sorted_weights.sum() / 2
        median_index = np.searchsorted(cumulative_weights, halfway)
        median = sorted_values[median_index]

        # std
        variance = np.average(
            (values - mean) ** 2,
            weights=weights_values
        )
        std = np.sqrt(variance)

        # range
        range = values.max() - values.min()

        rows.append([column, year, n_people, float(sum), float(mean), float(median), float(std), float(range)])

    return rows

def create_activity_summaries(dataframes):
    summaries = []

    for year, dataframe in dataframes.items():
        for level in ["detail", "1", "2"]:

            if level == "detail":
                # Only three-digit activities belong in this view.
                activities = keep_detailed_activities(dataframe)
            else:
                # Use ALL mapped activities, including broad codes
                # such as ha_3, when calculating group totals.
                activities = group_activities(
                    dataframe,
                    level=int(level),
                )

            summary_rows = activity_descriptions(
                activities,
                "weights",
                year,
            )

            summary = pl.DataFrame(
                summary_rows,
                schema=column_schema,
                orient="row",
            )

            summary = summary.with_columns(
                pl.lit(level).alias("level")
            )

            summaries.append(summary)

    return summaries


summaries = create_activity_summaries(dataframes_AG1529)

df_trends = df_trends.with_columns(

matches = []

for rows_inc in increases.iter_rows(named=True):
    for rows_dec in decreases.iter_rows(named=True):
        if abs(rows_inc["trend (i64)"] + rows_dec["trend (i64)"]) <= 1:
            matches.append(
                [rows_inc["activity_code"], 
                 rows_inc["activity_label"],
                 rows_dec["activity_code"], 
                 rows_dec["activity_label"],
                 rows_inc["trend (i64)"]
                ])

df_matches = pl.DataFrame(
    matches,
    schema=["increase: activity code", "increase: activity label", "decrease: activity code", "decrease: activity label", "trend +/- 1 (abs)"],
    orient="row"
)

by_activities = df_YBY_AG1529_described.filter(

by_activities

ranking_movers = []

for i in range(0, len(by_activities)):
    ranking_movers.append(["",0,0,0,0])

# counter for nested list based on per actvitiy
i = 0

for activitiy in by_activities:
    # set activity code as nested list start
    ranking_movers[i][0] = activitiy
    
    # rank - year 0
    rank_year0 = df_YBY_AG1529_described.filter(
            (pl.col("activity_code") == activitiy) &
            (pl.col("year") == years[0])
        )["ranking"].first()
    # set to last rank if none existing
    if rank_year0 is None:
        rank_year0 = df_YBY_AG1529_described["ranking"].max() + 1
    
    # rank - year 1
    rank_year1 = df_YBY_AG1529_described.filter(
                (pl.col("activity_code") == activitiy) &
                (pl.col("year") == years[1])
            )["ranking"].first()
    # set to last rank if none existing
    if rank_year1 is None:
        rank_year1 = df_YBY_AG1529_described["ranking"].max() + 1
    
    # rank - year 2
    rank_year2 = df_YBY_AG1529_described.filter(
                (pl.col("activity_code") == activitiy) &
                (pl.col("year") == years[2])
            )["ranking"].first()
    # set to last rank of if none existing
    if rank_year2 is None:
        rank_year2 = df_YBY_AG1529_described["ranking"].max() + 1
    
    # rank - year 3
    rank_year3 = df_YBY_AG1529_described.filter(
                (pl.col("activity_code") == activitiy) &
                (pl.col("year") == years[3])
            )["ranking"].first()
    # set to last rank if none existing
    if rank_year3 is None:
        rank_year3 = df_YBY_AG1529_described["ranking"].max() + 1
    
    # year ranks and set to 0
    # if None exists for previous year, use maximum rank of current year instead
    # if None exist for both years, set to 0
    if rank_year0 is not None and rank_year1 is not None:
        ranking_movers[i][1] = rank_year0 - rank_year1
    elif rank_year1 is not None:
        ranking_movers[i][1] = df_YBY_AG1529_described.filter(pl.col("year") == year[1])["ranking"].max() - rank_year1
    else:
        ranking_movers[i][1] = 0

    if rank_year1 is not None and rank_year2 is not None:
        ranking_movers[i][2] = rank_year1 - rank_year2
    elif rank_year2 is not None:
        ranking_movers[i][2] = df_YBY_AG1529_described.filter(pl.col("year") == year[2])["ranking"].max() - rank_year2
    else:
        ranking_movers[i][2] = 0
        
    if rank_year3 is not None and rank_year2 is not None:
        ranking_movers[i][3] = rank_year2 - rank_year3
    elif rank_year3 is not None:
        ranking_movers[i][3] = df_YBY_AG1529_described.filter(pl.col("year") == year[3])["ranking"].max() - rank_year3
    else:
        ranking_movers[i][3] = 0

    if rank_year3 is not None and rank_year0 is not None:
        ranking_movers[i][4] = rank_year0 - rank_year3
    elif rank_year3 is not None:
        ranking_movers[i][4] = df_YBY_AG1529_described.filter(pl.col("year") == year[1])["ranking"].max() - rank_year3
    else:
        ranking_movers[i][4] = 0
        
    i += 1
